\newcommand\tricolor[3]{
\begin{tikzpicture}
\fill[#1] (0,0) rectangle node[black] {#1} ++(1,2);
\fill[#2] (1,0) rectangle node[black] {#2} ++(1,2);
\fill[#3] (2,0) rectangle node[black] {#3} ++(1,2);
\end{tikzpicture}
}
